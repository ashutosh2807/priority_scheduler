@echo off
setlocal EnableExtensions

REM Windows Task Scheduler entry point: one locked cycle, then exit.
REM Schedule this file every 5 minutes. If a previous run is still executing
REM Oracle work, main.py reports a safe skip and exits successfully.
cd /d "%~dp0"

if not exist "%~dp0logs" mkdir "%~dp0logs"
echo [%date% %time%] Task Scheduler cycle started.>> "%~dp0logs\task_scheduler.log"

set "PYTHON_EXE=%~dp0venv\Scripts\python.exe"
call :validate_python
if not defined PYTHON_EXE (
    set "PYTHON_EXE=%~dp0..\.scheduler-venv\Scripts\python.exe"
    call :validate_python
)

if defined PYTHON_EXE (
    "%PYTHON_EXE%" "%~dp0main.py" --once >> "%~dp0logs\task_scheduler.log" 2>&1
) else (
    echo [%date% %time%] Python environment unavailable. Run ..\setup_scheduler_environment.bat first.>> "%~dp0logs\task_scheduler.log"
    exit /b 9009
)

set "EXIT_CODE=%ERRORLEVEL%"
echo [%date% %time%] Task Scheduler cycle finished with exit code %EXIT_CODE%.>> "%~dp0logs\task_scheduler.log"
exit /b %EXIT_CODE%

:validate_python
if not exist "%PYTHON_EXE%" (
    set "PYTHON_EXE="
    exit /b 0
)
"%PYTHON_EXE%" -c "import sys" >nul 2>&1
if errorlevel 1 set "PYTHON_EXE="
exit /b 0
